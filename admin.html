document.getElementById('m-login').classList.add('hidden'); document.getElementById('m-dash').classList.remove('hidden'); if (input) input.value = ''; initAdminDashboard(); logAudit('دخول الإدارة', 'تم تسجيل الدخول إلى بوابة المدير'); saveSecurityState(gateKey, st); alert(`تم تجاوز 10 محاولات! تم حظر الوصول لمدة ${banHours} ساعة.`); } else { saveSecurityState(gateKey, st); alert(`رمز المدير خطأ! المحاولات المتبقية: ${MAX_ATTEMPTS - st.failedAttempts} من ${MAX_ATTEMPTS}`); } } }
